Phase 1
localStorage — Remember Things
💡 Concept

localStorage saves data in the browser that persists after closing and reopening the page. Methods: localStorage.setItem('key','value') — save. localStorage.getItem('key') — retrieve. localStorage.removeItem('key') — delete. localStorage.clear() — delete all. Both key and value must be strings. This is how lesson progress is tracked throughout this Techbase HTML course!

index.html 
<!DOCTYPE html>
<html lang="en">
  <head><title>localStorage</title></head>
  <body>
    <h1>localStorage Demo</h1>
    <input type="text" id="nameInput" placeholder="Type your name">
    <button onclick="saveName()">Save</button>
    <button onclick="clearName()">Clear</button>
    <p id="greeting">Loading...</p>
    <script>
      function saveName() {
        const name = document.getElementById('nameInput').value;
        localStorage.setItem('username', name);
        showGreeting();
      }
      function clearName() {
        localStorage.removeItem('username');
        document.getElementById('greeting').textContent = 'Name cleared!';
      }
      function showGreeting() {
        const saved = localStorage.getItem('username');
        const el = document.getElementById('greeting');
        if (saved) { el.textContent = 'Welcome back, ' + saved + '! 👋';
                     el.style.color = '#10b981'; }
        else        { el.textContent = 'No name saved yet.'; }
      }
      showGreeting();
    </script>
  </body>
</html>

Phase 2
Geolocation API
💡 Concept

The Geolocation API asks permission to access the user's location. Call navigator.geolocation.getCurrentPosition(successFn, errorFn). Success gives position.coords.latitude and position.coords.longitude. The browser always shows a permission prompt — the user can deny it. Works great on mobile phones!

index.html 
<!DOCTYPE html>
<html lang="en">
  <head><title>Geolocation</title></head>
  <body>
    <h1>My Location</h1>
    <button onclick="getLocation()">📍 Find My Location</button>
    <p id="out">Click to request location.</p>
    <script>
      function getLocation() {
        const out = document.getElementById('out');
        if (!navigator.geolocation) {
          out.textContent = 'Not supported.'; return;
        }
        out.textContent = '⏳ Getting location...';
        navigator.geolocation.getCurrentPosition(
          function(pos) {
            const lat = pos.coords.latitude.toFixed(4);
            const lon = pos.coords.longitude.toFixed(4);
            out.innerHTML = '📍 Lat: <strong>' + lat +
                            '</strong>, Lon: <strong>' + lon + '</strong><br>' +
                            '<a href="https://maps.google.com/?q=' +
                            lat + ',' + lon + '" target="_blank" ' +
                            'style="color:#ea4f21;">Open in Google Maps ↗</a>';
          },
          function() { out.textContent = '❌ Location denied.'; }
        );
      }
    </script>
  </body>
</html>

Phase 3
Dark Mode, Online Status and Clipboard
💡 Concept

Three more handy APIs: window.matchMedia('(prefers-color-scheme: dark)').matches detects if the user has dark mode on their device. navigator.onLine returns true/false for internet connection. navigator.clipboard.writeText(text) copies text to clipboard. Use CSS @media (prefers-color-scheme: dark) to automatically apply dark styles for dark-mode users — no JavaScript needed!

index.html 
<!DOCTYPE html>
<html lang="en">
  <head>
    <title>More APIs</title>
    <style>
      /* Auto dark mode — activates if OS dark mode is on! */
      @media (prefers-color-scheme: dark) {
        body { background:#0f172a; color:#f1f5f9; }
      }
      button { padding:10px 20px; margin:6px; border-radius:8px;
               border:none; background:#ea4f21; color:white;
               font-weight:bold; cursor:pointer; }
    </style>
  </head>
  <body>
    <h1>Browser APIs</h1>
    <p>Internet: <strong id="status"></strong></p>
    <button onclick="copyLink()">📋 Copy Page URL</button>
    <span id="copy-msg"></span>
    <script>
      function updateStatus() {
        document.getElementById('status').textContent =
          navigator.onLine ? '✅ Online' : '❌ Offline';
      }
      window.addEventListener('online',  updateStatus);
      window.addEventListener('offline', updateStatus);
      updateStatus();
      function copyLink() {
        navigator.clipboard.writeText(window.location.href).then(() => {
          const m = document.getElementById('copy-msg');
          m.textContent = ' ✅ Copied!';
          setTimeout(()=> m.textContent='', 2000);
        });
      }
    </script>
  </body>
</html>
