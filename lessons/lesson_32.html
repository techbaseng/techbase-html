Phase 1
SVG — Scalable Vector Graphics
💡 Concept

SVG draws shapes using HTML-like tags. Shapes scale to any size without blurring — perfect for logos and icons. Basic shapes: <rect> (rectangle with optional rx for rounded corners), <circle> (cx/cy=centre, r=radius), <line>, <polygon> (points), <text>. Attributes: fill (colour), stroke (outline colour), stroke-width.

index.html 
<!DOCTYPE html>
<html lang="en">
  <head><title>SVG Shapes</title></head>
  <body>
    <h1>SVG Drawing</h1>
    <svg width="400" height="300"
         style="border:1px solid #e2e8f0; border-radius:10px; background:#f8fafc;">
      <!-- Rectangle with rounded corners -->
      <rect x="20" y="20" width="120" height="80"
            fill="#ea4f21" rx="10"/>
      <!-- Circle -->
      <circle cx="260" cy="60" r="50" fill="#0284c7"/>
      <!-- Triangle -->
      <polygon points="300,120 380,270 220,270" fill="#d97706"/>
      <!-- Ellipse -->
      <ellipse cx="110" cy="210" rx="90" ry="50" fill="#10b981"/>
      <!-- Line -->
      <line x1="20" y1="290" x2="380" y2="290"
            stroke="#7c3aed" stroke-width="3"/>
      <!-- Text -->
      <text x="20" y="285" fill="white"
            font-family="Arial" font-size="14" font-weight="bold">
        SVG shapes!
      </text>
    </svg>
  </body>
</html>

Phase 2
HTML Canvas with JavaScript
💡 Concept

The <canvas> element is a drawing surface controlled with JavaScript. Get the context: const ctx = canvas.getContext('2d'). Then draw: fillRect(x,y,w,h), beginPath(), arc(x,y,r,0,2*Math.PI) for circles, fillText(text,x,y). Canvas is pixel-based — better than SVG for games and animations.

index.html 
<!DOCTYPE html>
<html lang="en">
  <head><title>Canvas</title></head>
  <body>
    <h1>Canvas Drawing</h1>
    <canvas id="myCanvas" width="500" height="300"
            style="border:1px solid #e2e8f0; border-radius:10px; display:block;"></canvas>
    <script>
      const canvas = document.getElementById('myCanvas');
      const ctx = canvas.getContext('2d');
      // Orange rectangle
      ctx.fillStyle = '#ea4f21';
      ctx.fillRect(20, 20, 150, 80);
      // Blue circle
      ctx.fillStyle = '#0284c7';
      ctx.beginPath();
      ctx.arc(320, 80, 60, 0, 2 * Math.PI);
      ctx.fill();
      // Green triangle (path)
      ctx.fillStyle = '#10b981';
      ctx.beginPath();
      ctx.moveTo(100, 240);
      ctx.lineTo(200, 140);
      ctx.lineTo(300, 240);
      ctx.closePath();
      ctx.fill();
      // Text
      ctx.fillStyle = '#0f172a';
      ctx.font = 'bold 18px Arial';
      ctx.fillText('Canvas!', 20, 280);
    </script>
  </body>
</html>

Phase 3
SVG vs Canvas
💡 Concept

SVG is best for: logos, icons, interactive graphics (CSS hover works!), and accessible content (screen readers can read SVG text). Canvas is best for: pixel-level control, smooth animations (games), large numbers of moving objects, and image processing. Most modern sites use SVG for icons and Canvas for games or live charts.

index.html 
<!DOCTYPE html>
<html lang="en">
  <head>
    <title>SVG vs Canvas</title>
    <style>
      .demos { display:flex; gap:20px; flex-wrap:wrap; }
      .demo  { background:white; border-radius:12px; padding:16px;
               border:1.5px solid #e2e8f0; }
    </style>
  </head>
  <body>
    <h1>SVG vs Canvas</h1>
    <div class="demos">
      <div class="demo">
        <h3>SVG — hover to grow!</h3>
        <svg width="200" height="200">
          <style>circle { transition:r .3s; } circle:hover { r:75; }</style>
          <circle cx="100" cy="100" r="60" fill="#ea4f21" style="cursor:pointer;"/>
          <text x="100" y="108" text-anchor="middle"
                fill="white" font-family="Arial" font-weight="bold">SVG</text>
        </svg>
      </div>
      <div class="demo">
        <h3>Canvas — animated!</h3>
        <canvas id="c" width="200" height="200"></canvas>
        <script>
          const c=document.getElementById('c'), x=c.getContext('2d');
          let a=0;
          function draw(){
            x.clearRect(0,0,200,200);
            x.fillStyle='#0f172a'; x.fillRect(0,0,200,200);
            x.fillStyle='#0284c7';
            x.beginPath();
            x.arc(100+Math.cos(a)*40, 100+Math.sin(a)*40, 20, 0, 2*Math.PI);
            x.fill();
            a+=0.03; requestAnimationFrame(draw);
          }
          draw();
        </script>
      </div>
    </div>
  </body>
</html>

My Graphics Page

<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>My Graphics Page — Lesson 32</title>
    <style>
      * { box-sizing:border-box; }
      body { font-family:Arial,sans-serif; background:#f1f5f9;
             padding:30px; max-width:960px; margin:0 auto; }
      h1 { color:#ea4f21; text-align:center; }
      .grid { display:flex; gap:24px; flex-wrap:wrap; margin-top:30px; }
      .panel { flex:1; min-width:300px; background:white;
               border-radius:16px; padding:24px;
               box-shadow:0 2px 10px rgba(0,0,0,.07); }
      .panel h2 { color:#0f172a; margin:0 0 16px; font-size:1.1rem; }
      .panel p  { color:#64748b; font-size:13px; margin-top:12px; }
      canvas, svg { border-radius:10px; display:block; width:100%; }
    </style>
  </head>
  <body>
    <h1>🎨 My HTML Graphics Page</h1>
    <div class="grid">
      <div class="panel">
        <h2>🔷 SVG — Landscape</h2>
        <svg viewBox="0 0 300 200" style="background:#dbeafe;">
          <rect width="300" height="200" fill="#dbeafe"/>
          <circle cx="250" cy="40" r="25" fill="#fbbf24"/>
          <polygon points="0,200 100,80 200,200"  fill="#94a3b8"/>
          <polygon points="80,200 190,60 300,200" fill="#64748b"/>
          <rect x="0" y="170" width="300" height="30" fill="#10b981"/>
          <rect x="30" y="150" width="8" height="22" fill="#78350f"/>
          <polygon points="34,110 18,152 50,152" fill="#166534"/>
          <text x="6" y="196" font-family="Arial" font-size="10"
                fill="rgba(0,0,0,.4)">SVG Landscape — scales perfectly!</text>
        </svg>
        <p>✅ Scales without blurring | ✅ CSS hover works | ✅ Screen-reader friendly</p>
      </div>
      <div class="panel" style="background:#0f172a;">
        <h2 style="color:#f1f5f9;">🔶 Canvas — Particle Animation</h2>
        <canvas id="myCanvas" width="400" height="220"></canvas>
        <p style="color:#475569;">✅ Smooth 60fps animation | ✅ Game-ready | ✅ Pixel control</p>
      </div>
    </div>
    <script>
      const canvas=document.getElementById('myCanvas'), ctx=canvas.getContext('2d');
      const W=canvas.width, H=canvas.height;
      const pts=Array.from({length:50},()=>({
        x:Math.random()*W, y:Math.random()*H,
        vx:(Math.random()-.5), vy:(Math.random()-.5),
        r:Math.random()*3+1,
        c:['#ea4f21','#0284c7','#10b981','#7c3aed','#fbbf24'][Math.floor(Math.random()*5)]
      }));
      function draw(){
        ctx.fillStyle='rgba(15,23,42,.2)'; ctx.fillRect(0,0,W,H);
        pts.forEach(p=>{
          p.x+=p.vx; p.y+=p.vy;
          if(p.x<0||p.x>W) p.vx*=-1;
          if(p.y<0||p.y>H) p.vy*=-1;
          ctx.fillStyle=p.c;
          ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,2*Math.PI); ctx.fill();
        });
        requestAnimationFrame(draw);
      }
      draw();
    </script>
  </body>
</html>
